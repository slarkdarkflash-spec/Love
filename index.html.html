function showLoading(){

document.getElementById("quiz").classList.add("hidden");
document.getElementById("loading").classList.remove("hidden");

let bar=document.getElementById("bar");
let text=document.getElementById("loadText");

let percent=0;

let timer=setInterval(()=>{

percent+=10;
bar.style.width=percent+"%";

if(percent==20)
text.innerHTML="در حال بررسی جواب‌ها... 🔍";

if(percent==50)
text.innerHTML="در حال اندازه‌گیری میزان علاقه... ❤️";

if(percent==80)
text.innerHTML="تقریباً تمام شد... 😍";

if(percent>=100){

clearInterval(timer);

setTimeout(showResult,1000);

}

},400);

}


function showResult(){

document.getElementById("loading").classList.add("hidden");
document.getElementById("result").classList.remove("hidden");

createHearts();

let message=
"نتیجه آزمایش عشق:<br><br>"+
"❤️ وضعیت: عاشقانه<br>"+
"💯 میزان علاقه: 100%<br><br>"+
"جواب‌هایی که دادی مهم نبود...<br>"+
"چون من از اول جواب قلبت رو می‌دونستم ❤️<br><br>"+
"فقط می‌خواستم یه بهونه داشته باشم که بگم...<br><br>"+
"<b>دوستت دارم تا بی‌نهایت 🥹❤️</b>";

typeText(message);

}


function typeText(text){

let box=document.getElementById("finalText");

let i=0;
box.innerHTML="";

let timer=setInterval(()=>{

box.innerHTML=text.substring(0,i);

i++;

if(i>text.length){

clearInterval(timer);

}

},40);

}



function createHearts(){

for(let i=0;i<40;i++){

let heart=document.createElement("div");

heart.className="heart";
heart.innerHTML="❤️";

heart.style.left=Math.random()*100+"%";

heart.style.animationDuration=
(4+Math.random()*5)+"s";

heart.style.fontSize=
(15+Math.random()*30)+"px";

document.body.appendChild(heart);

}

}
